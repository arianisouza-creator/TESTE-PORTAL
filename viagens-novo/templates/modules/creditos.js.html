document.getElementById('formCredito').addEventListener('submit', async (e) => { e.preventDefault(); const fd = Object.fromEntries(new FormData(e.target).entries()); const id = genId('cred'); await apiFetch('/rest/v1/passagens_creditos', {method:'POST', body:JSON.stringify({id, data:fd})}); e.target.reset(); loadCreditos(); }); async function loadCreditos(){ const rows = await apiFetch('/rest/v1/passagens_creditos'); document.querySelector('#tabelaCreditos tbody').innerHTML = rows.map(r => { const d=r.data; return ` ${d.companhia||'-'}${d.localizador||'-'}${d.valor?fmtMoney(d.valor):'-'} ${fmtDate(d.validade)}${d.status||'-'}`; }).join('') || `Nenhum crédito cadastrado ainda.`; }
